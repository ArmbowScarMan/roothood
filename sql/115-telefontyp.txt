-- 115: telefontyp i statistiken (bara ett ord per användare: android, iphone, dator eller annat)
alter table public.presence add column if not exists device text;
alter table public.presence drop constraint if exists presence_device_chk;
alter table public.presence add constraint presence_device_chk check (device is null or device in ('android', 'iphone', 'dator', 'annat'));

drop function if exists public.presence_ping(boolean);
create or replace function public.presence_ping(p_action boolean default false, p_device text default null)
returns void language plpgsql security definer set search_path to 'public' as $$
declare d text := case when p_device in ('android', 'iphone', 'dator', 'annat') then p_device else null end;
begin
  if auth.uid() is null then return; end if;
  insert into public.presence (user_id, last_seen, last_action, device)
  values (auth.uid(), now(), case when p_action then now() else now() - interval '1 day' end, d)
  on conflict (user_id) do update
    set last_seen = now(),
        last_action = case when p_action then now() else public.presence.last_action end,
        device = coalesce(d, public.presence.device);
end $$;

drop function if exists public.mod_usage_users(text, integer);
create or replace function public.mod_usage_users(p_feature text default '', p_days integer default 30)
returns table(user_id uuid, display_name text, uses integer, features integer, last_day date, device text)
language plpgsql stable security definer set search_path to 'public' as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  return query select u.user_id, p.display_name, sum(u.uses)::int, count(distinct u.feature)::int, max(u.day), pr.device
    from usage_daily u left join profiles p on p.id = u.user_id left join presence pr on pr.user_id = u.user_id
    where u.day > current_date - greatest(1, least(p_days, 365)) and (coalesce(p_feature, '') = '' or u.feature = p_feature)
    group by u.user_id, p.display_name, pr.device order by 3 desc limit 100;
end $$;

drop function if exists public.mod_usage_users_h(text, integer);
create or replace function public.mod_usage_users_h(p_feature text default '', p_hours integer default 24)
returns table(user_id uuid, display_name text, uses integer, features integer, last_day date, device text)
language plpgsql stable security definer set search_path to 'public' as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  return query select u.user_id, p.display_name, sum(u.uses)::int, count(distinct u.feature)::int, max(u.hour)::date, pr.device
    from usage_hourly u left join profiles p on p.id = u.user_id left join presence pr on pr.user_id = u.user_id
    where u.hour > now() - make_interval(hours => greatest(1, least(p_hours, 168))) and (coalesce(p_feature, '') = '' or u.feature = p_feature)
    group by u.user_id, p.display_name, pr.device order by 3 desc limit 100;
end $$;
