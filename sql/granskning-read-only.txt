-- Granskning: bara läsning, ändrar ingenting. Klistra in en fråga i taget i Supabase SQL-editorn.

-- 1. Tabeller i public som SAKNAR radsäkerhet (RLS). Väntat resultat: tom lista,
--    eller bara tabeller du vet är avsiktligt öppna.
select c.relname as tabell
from pg_class c join pg_namespace n on n.oid = c.relnamespace
where n.nspname = 'public' and c.relkind = 'r' and not c.relrowsecurity
order by 1;

-- 2. Alla regler (policies) i public. Skicka resultatet till chatten.
select tablename, policyname, cmd, qual, with_check
from pg_policies where schemaname = 'public'
order by tablename, cmd;

-- 3. Funktioner med SECURITY DEFINER. Väntat: search_path satt i proconfig.
select p.proname, p.proconfig
from pg_proc p join pg_namespace n on n.oid = p.pronamespace
where n.nspname = 'public' and p.prosecdef
order by 1;
