-- Roothood steg 029: privata meddelanden mellan vänner, och ett valfritt meddelande med vänförfrågan.
-- Kräver steg 027 och 028. Det gör inget om du kör den två gånger.

alter table public.friend_requests add column if not exists message text;
alter table public.friend_requests drop constraint if exists friend_requests_message_len;
alter table public.friend_requests add constraint friend_requests_message_len check (message is null or char_length(message) <= 300);

create table if not exists public.messages (
  id bigint generated by default as identity primary key,
  from_user uuid not null default auth.uid() references auth.users on delete cascade,
  to_user uuid not null references auth.users on delete cascade,
  body text not null check (char_length(body) between 1 and 1000),
  created_at timestamptz not null default now(),
  read_at timestamptz,
  check (from_user <> to_user)
);
create index if not exists messages_to_idx on public.messages (to_user, created_at desc);
create index if not exists messages_from_idx on public.messages (from_user, created_at desc);
alter table public.messages enable row level security;
drop policy if exists "Läs egna meddelanden" on public.messages;
create policy "Läs egna meddelanden" on public.messages for select using (auth.uid() in (from_user, to_user));
drop policy if exists "Skicka till vän" on public.messages;
create policy "Skicka till vän" on public.messages for insert with check (auth.uid() = from_user and public.are_friends(from_user, to_user));

create or replace function public.no_blocked_messages()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if exists (select 1 from blocks b where (b.blocker = new.from_user and b.blocked = new.to_user) or (b.blocker = new.to_user and b.blocked = new.from_user)) then
    raise exception 'BLOCKED';
  end if;
  return new;
end $$;
drop trigger if exists no_blocked_messages_t on public.messages;
create trigger no_blocked_messages_t before insert on public.messages for each row execute function public.no_blocked_messages();

-- Samtalslista: senaste meddelandet och antal olästa per person
create or replace function public.inbox()
returns table (other uuid, name text, last_body text, last_at timestamptz, last_mine boolean, unread int)
language sql stable security definer set search_path = public as $$
  select o.other, p.display_name, m.body, m.created_at, (m.from_user = auth.uid()),
    (select count(*)::int from messages x where x.from_user = o.other and x.to_user = auth.uid() and x.read_at is null)
  from (select distinct case when from_user = auth.uid() then to_user else from_user end as other from messages where auth.uid() in (from_user, to_user)) o
  join lateral (select * from messages mm where (mm.from_user = auth.uid() and mm.to_user = o.other) or (mm.from_user = o.other and mm.to_user = auth.uid()) order by mm.created_at desc limit 1) m on true
  left join profiles p on p.id = o.other
  where not exists (select 1 from blocks b where (b.blocker = auth.uid() and b.blocked = o.other) or (b.blocker = o.other and b.blocked = auth.uid()))
  order by m.created_at desc;
$$;
grant execute on function public.inbox() to authenticated;

create or replace function public.mark_read(p_other uuid)
returns void language sql security definer set search_path = public as $$
  update messages set read_at = now() where from_user = p_other and to_user = auth.uid() and read_at is null;
$$;
grant execute on function public.mark_read(uuid) to authenticated;
