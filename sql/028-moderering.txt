-- 1. Roller
alter table public.profiles add column if not exists role text not null default 'user';
alter table public.profiles drop constraint if exists profiles_role_check;
alter table public.profiles add constraint profiles_role_check check (role in ('user', 'mod', 'admin'));

create or replace function public.my_role()
returns text language sql stable security definer set search_path = public as $$
  select coalesce((select role from profiles where id = auth.uid()), 'user');
$$;
create or replace function public.is_mod()
returns boolean language sql stable security definer set search_path = public as $$
  select public.my_role() in ('mod', 'admin');
$$;
grant execute on function public.my_role() to authenticated;
grant execute on function public.is_mod() to authenticated;

-- Ingen kan ge sig själv en roll. Bara en admin (eller SQL Editor) kan ändra roller.
create or replace function public.protect_role()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if auth.uid() is not null then
    if tg_op = 'INSERT' then new.role := 'user';
    elsif new.role is distinct from old.role and public.my_role() <> 'admin' then new.role := old.role; end if;
  end if;
  return new;
end $$;
drop trigger if exists protect_role_t on public.profiles;
create trigger protect_role_t before insert or update on public.profiles for each row execute function public.protect_role();

-- 2. Dolda inlägg och svar, och status på rapporter
alter table public.posts add column if not exists hidden boolean not null default false;
alter table public.answers add column if not exists hidden boolean not null default false;
alter table public.reports add column if not exists id bigint generated by default as identity;
alter table public.reports add column if not exists created_at timestamptz not null default now();
alter table public.reports add column if not exists status text not null default 'open';
alter table public.reports add column if not exists handled_by uuid;
alter table public.reports add column if not exists handled_at timestamptz;

-- 3. Dölj automatiskt när tre olika personer rapporterat samma inlägg eller svar
create or replace function public.reports_auto()
returns trigger language plpgsql security definer set search_path = public as $$
declare n int;
begin
  if new.post_id is not null then
    select count(distinct reporter_id) into n from reports where post_id = new.post_id;
    if n >= 3 then update posts set hidden = true where id = new.post_id; end if;
  elsif new.answer_id is not null then
    select count(distinct reporter_id) into n from reports where answer_id = new.answer_id;
    if n >= 3 then update answers set hidden = true where id = new.answer_id; end if;
  end if;
  return new;
end $$;
drop trigger if exists reports_auto_t on public.reports;
create trigger reports_auto_t after insert on public.reports for each row execute function public.reports_auto();

-- 4. Moderatorernas funktioner
create or replace function public.mod_reports(p_status text default 'open')
returns table (id bigint, kind text, target text, reason text, created_at timestamptz, reporter text, author_id uuid, author text, excerpt text, hidden boolean, n int, status text)
language plpgsql stable security definer set search_path = public as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  return query
  select r.id,
    (case when r.post_id is not null then 'post' when r.answer_id is not null then 'answer' else 'photo' end)::text,
    coalesce(r.post_id::text, r.answer_id::text, r.photo_url),
    r.reason, r.created_at, rp.display_name,
    coalesce(po.user_id, an.user_id), ap.display_name,
    left(coalesce(po.title || ' ', '') || coalesce(po.body, an.body, ''), 200),
    coalesce(po.hidden, an.hidden, false),
    (select count(*)::int from reports r2 where r2.post_id is not distinct from r.post_id and r2.answer_id is not distinct from r.answer_id and r2.photo_url is not distinct from r.photo_url),
    r.status
  from reports r
  left join profiles rp on rp.id = r.reporter_id
  left join posts po on po.id = r.post_id
  left join answers an on an.id = r.answer_id
  left join profiles ap on ap.id = coalesce(po.user_id, an.user_id)
  where p_status = 'all' or r.status = p_status
  order by r.created_at desc limit 100;
end $$;
grant execute on function public.mod_reports(text) to authenticated;

create or replace function public.mod_act(p_report bigint, p_action text)
returns void language plpgsql security definer set search_path = public as $$
declare r reports%rowtype;
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  select * into r from reports where id = p_report;
  if not found then return; end if;
  if p_action in ('hide', 'unhide', 'keep') then
    if r.post_id is not null then update posts set hidden = (p_action = 'hide') where id = r.post_id;
    elsif r.answer_id is not null then update answers set hidden = (p_action = 'hide') where id = r.answer_id;
    elsif r.photo_url is not null and p_action = 'hide' then update user_plants set show_own = false where photo_url = r.photo_url; end if;
  end if;
  update reports set status = case p_action when 'hide' then 'hidden' else 'kept' end, handled_by = auth.uid(), handled_at = now()
  where post_id is not distinct from r.post_id and answer_id is not distinct from r.answer_id and photo_url is not distinct from r.photo_url and status = 'open';
end $$;
grant execute on function public.mod_act(bigint, text) to authenticated;

create or replace function public.mod_open_count()
returns int language sql stable security definer set search_path = public as $$
  select case when public.is_mod() then (select count(*)::int from (select distinct post_id, answer_id, photo_url from reports where status = 'open') x) else 0 end;
$$;
grant execute on function public.mod_open_count() to authenticated;

create or replace function public.mod_feed()
returns table (kind text, id bigint, author_id uuid, author text, excerpt text, created_at timestamptz, hidden boolean)
language plpgsql stable security definer set search_path = public as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  return query
  select * from (
    select 'post'::text, p.id::bigint, p.user_id, pr.display_name, left(coalesce(p.title || ' ', '') || coalesce(p.body, ''), 160), p.created_at, p.hidden
    from posts p left join profiles pr on pr.id = p.user_id
    union all
    select 'answer'::text, a.id::bigint, a.user_id, pr.display_name, left(coalesce(a.body, ''), 160), a.created_at, a.hidden
    from answers a left join profiles pr on pr.id = a.user_id
  ) x order by 6 desc limit 60;
end $$;
grant execute on function public.mod_feed() to authenticated;

create or replace function public.mod_hide(p_kind text, p_id bigint, p_hide boolean)
returns void language plpgsql security definer set search_path = public as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  if p_kind = 'post' then update posts set hidden = p_hide where id = p_id;
  elsif p_kind = 'answer' then update answers set hidden = p_hide where id = p_id; end if;
end $$;
grant execute on function public.mod_hide(text, bigint, boolean) to authenticated;

-- 5. Team: bara admin lägger till och tar bort moderatorer
create or replace function public.list_staff()
returns table (id uuid, display_name text, role text)
language plpgsql stable security definer set search_path = public as $$
begin
  if not public.is_mod() then raise exception 'NOT_MOD'; end if;
  return query select p.id, p.display_name, p.role from profiles p where p.role in ('mod', 'admin') order by p.role, p.display_name;
end $$;
grant execute on function public.list_staff() to authenticated;

create or replace function public.set_role(p_user uuid, p_role text)
returns void language plpgsql security definer set search_path = public as $$
begin
  if public.my_role() <> 'admin' then raise exception 'NOT_ADMIN'; end if;
  if p_role not in ('user', 'mod') then raise exception 'BAD_ROLE'; end if;
  if p_user = auth.uid() then raise exception 'SELF'; end if;
  update profiles set role = p_role where id = p_user and role <> 'admin';
end $$;
grant execute on function public.set_role(uuid, text) to authenticated;

-- 6. Blockera användare
create table if not exists public.blocks (
  blocker uuid not null default auth.uid() references auth.users on delete cascade,
  blocked uuid not null references auth.users on delete cascade,
  created_at timestamptz not null default now(),
  primary key (blocker, blocked),
  check (blocker <> blocked)
);
alter table public.blocks enable row level security;
drop policy if exists "Läs egna blockeringar" on public.blocks;
create policy "Läs egna blockeringar" on public.blocks for select using (auth.uid() = blocker);
drop policy if exists "Blockera själv" on public.blocks;
create policy "Blockera själv" on public.blocks for insert with check (auth.uid() = blocker);
drop policy if exists "Avblockera själv" on public.blocks;
create policy "Avblockera själv" on public.blocks for delete using (auth.uid() = blocker);

-- Vid blockering tas vänskap och följning bort
create or replace function public.on_block()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  delete from friend_requests where (from_user = new.blocker and to_user = new.blocked) or (from_user = new.blocked and to_user = new.blocker);
  delete from follows where (follower = new.blocker and followee = new.blocked) or (follower = new.blocked and followee = new.blocker);
  return new;
end $$;
drop trigger if exists on_block_t on public.blocks;
create trigger on_block_t after insert on public.blocks for each row execute function public.on_block();

-- Man kan inte skicka vänförfrågan till den som blockerat en
create or replace function public.no_blocked_requests()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if exists (select 1 from blocks b where (b.blocker = new.from_user and b.blocked = new.to_user) or (b.blocker = new.to_user and b.blocked = new.from_user)) then
    raise exception 'BLOCKED';
  end if;
  return new;
end $$;
drop trigger if exists no_blocked_requests_t on public.friend_requests;
create trigger no_blocked_requests_t before insert on public.friend_requests for each row execute function public.no_blocked_requests();

-- Blockerade syns inte i sökningen
create or replace function public.find_users(q text default '')
returns table (id uuid, display_name text, city text, interests text[], plants int, posts int, followers int)
language sql stable security definer set search_path = public as $$
  select p.id, p.display_name, case when can_see(p.id, 'city') then p.city end, case when can_see(p.id, 'interests') then p.interests else '{}'::text[] end,
    (select count(*) from user_plants u where u.user_id = p.id)::int,
    (select count(*) from posts x where x.user_id = p.id and not coalesce(x.hidden, false))::int,
    (select count(*) from follows f where f.followee = p.id)::int
  from profiles p
  where p.id is distinct from auth.uid()
    and not exists (select 1 from blocks b where (b.blocker = auth.uid() and b.blocked = p.id) or (b.blocker = p.id and b.blocked = auth.uid()))
    and (coalesce(q, '') = '' or p.display_name ilike '%' || q || '%' or (can_see(p.id, 'city') and p.city ilike '%' || q || '%'))
  order by (can_see(p.id, 'city') and lower(coalesce(p.city, '')) = lower(coalesce((select city from profiles where id = auth.uid()), '#'))) desc,
    (select count(*) from follows f where f.followee = p.id) desc, p.display_name
  limit 40;
$$;
grant execute on function public.find_users(text) to authenticated;
