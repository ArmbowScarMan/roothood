-- Roothood steg 027: vänförfrågningar och val för vad andra ser på din profil.
-- Kör en gång i Supabase. Det gör inget om du kör den två gånger.

-- 1. Synlighet: vad ser andra? (all = alla, friends = bara vänner, none = ingen)
alter table public.profiles add column if not exists vis jsonb not null default '{}'::jsonb;

-- 2. Mina djur (finns redan om steg 024 är körd)
create table if not exists public.pets (
  owner uuid primary key default auth.uid() references auth.users on delete cascade,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);
alter table public.pets enable row level security;
drop policy if exists "Egna djur läs" on public.pets;
create policy "Egna djur läs" on public.pets for select using (auth.uid() = owner);
drop policy if exists "Egna djur ny" on public.pets;
create policy "Egna djur ny" on public.pets for insert with check (auth.uid() = owner);
drop policy if exists "Egna djur ändra" on public.pets;
create policy "Egna djur ändra" on public.pets for update using (auth.uid() = owner) with check (auth.uid() = owner);

-- 3. Vänförfrågningar
create table if not exists public.friend_requests (
  id bigint generated by default as identity primary key,
  from_user uuid not null default auth.uid() references auth.users on delete cascade,
  to_user uuid not null references auth.users on delete cascade,
  status text not null default 'pending' check (status in ('pending', 'accepted')),
  created_at timestamptz not null default now(),
  responded_at timestamptz,
  check (from_user <> to_user)
);
create unique index if not exists friend_pair on public.friend_requests (least(from_user, to_user), greatest(from_user, to_user));
alter table public.friend_requests enable row level security;
drop policy if exists "Läs egna förfrågningar" on public.friend_requests;
create policy "Läs egna förfrågningar" on public.friend_requests for select using (auth.uid() in (from_user, to_user));
drop policy if exists "Skicka förfrågan" on public.friend_requests;
create policy "Skicka förfrågan" on public.friend_requests for insert with check (auth.uid() = from_user and status = 'pending');
drop policy if exists "Acceptera förfrågan" on public.friend_requests;
create policy "Acceptera förfrågan" on public.friend_requests for update using (auth.uid() = to_user) with check (auth.uid() = to_user and status = 'accepted');
drop policy if exists "Neka eller ta bort" on public.friend_requests;
create policy "Neka eller ta bort" on public.friend_requests for delete using (auth.uid() in (from_user, to_user));

create or replace function public.are_friends(a uuid, b uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from friend_requests f where f.status = 'accepted'
    and ((f.from_user = a and f.to_user = b) or (f.from_user = b and f.to_user = a)));
$$;
grant execute on function public.are_friends(uuid, uuid) to authenticated;

create or replace function public.can_see(p_owner uuid, p_item text)
returns boolean language sql stable security definer set search_path = public as $$
  select p_owner = auth.uid() or case coalesce(
      (select vis ->> p_item from profiles where id = p_owner),
      case p_item when 'pets' then 'friends'
        when 'plants' then (case when coalesce((select share_photos from profiles where id = p_owner), true) then 'all' else 'none' end)
        else 'all' end)
    when 'all' then true
    when 'friends' then public.are_friends(p_owner, auth.uid())
    else false end;
$$;
grant execute on function public.can_see(uuid, text) to authenticated;

-- 4. Sök användare (dold ort syns inte)
create or replace function public.find_users(q text default '')
returns table (id uuid, display_name text, city text, interests text[], plants int, posts int, followers int)
language sql stable security definer set search_path = public as $$
  select p.id, p.display_name, case when can_see(p.id, 'city') then p.city end, case when can_see(p.id, 'interests') then p.interests else '{}'::text[] end,
    (select count(*) from user_plants u where u.user_id = p.id)::int,
    (select count(*) from posts x where x.user_id = p.id and not coalesce(x.hidden, false))::int,
    (select count(*) from follows f where f.followee = p.id)::int
  from profiles p
  where p.id is distinct from auth.uid()
    and (coalesce(q, '') = '' or p.display_name ilike '%' || q || '%' or (can_see(p.id, 'city') and p.city ilike '%' || q || '%'))
  order by (can_see(p.id, 'city') and lower(coalesce(p.city, '')) = lower(coalesce((select city from profiles where id = auth.uid()), '#'))) desc,
    (select count(*) from follows f where f.followee = p.id) desc, p.display_name
  limit 40;
$$;
grant execute on function public.find_users(text) to authenticated;

-- 5. Profilkort med vänstatus och synlighet
drop function if exists public.user_card(uuid);
create function public.user_card(p_id uuid)
returns table (id uuid, display_name text, city text, interests text[], plants int, posts int, followers int, following int, share boolean, plant_list jsonb,
  friends int, friend_status text, friend_req bigint, pets jsonb, hidden text[])
language sql stable security definer set search_path = public as $$
  select p.id, p.display_name,
    case when can_see(p.id, 'city') then p.city end,
    case when can_see(p.id, 'interests') then p.interests else '{}'::text[] end,
    (select count(*) from user_plants u where u.user_id = p.id)::int,
    (select count(*) from posts x where x.user_id = p.id and not coalesce(x.hidden, false))::int,
    (select count(*) from follows f where f.followee = p.id)::int,
    (select count(*) from follows f where f.follower = p.id)::int,
    can_see(p.id, 'plants'),
    case when can_see(p.id, 'plants') then (
      select jsonb_agg(jsonb_build_object('plant_id', up.plant_id, 'name', coalesce(pl.swedish_name, up.custom_name),
        'photo', case when array_length(up.photos, 1) > 0 and up.photos[array_length(up.photos, 1)] like 'https://%' then up.photos[array_length(up.photos, 1)] end) order by up.created_at desc)
      from user_plants up left join plants pl on pl.id = up.plant_id where up.user_id = p.id)
    end,
    (select count(*) from friend_requests f where f.status = 'accepted' and p.id in (f.from_user, f.to_user))::int,
    case when p.id = auth.uid() then 'self'
      when public.are_friends(p.id, auth.uid()) then 'friends'
      when exists (select 1 from friend_requests f where f.status = 'pending' and f.from_user = auth.uid() and f.to_user = p.id) then 'pending_out'
      when exists (select 1 from friend_requests f where f.status = 'pending' and f.from_user = p.id and f.to_user = auth.uid()) then 'pending_in'
      else 'none' end,
    (select f.id from friend_requests f where f.status = 'pending' and f.from_user = p.id and f.to_user = auth.uid() limit 1),
    case when can_see(p.id, 'pets') then (select d.data -> 'list' from pets d where d.owner = p.id) end,
    array_remove(array[
      case when not can_see(p.id, 'city') then 'city' end,
      case when not can_see(p.id, 'interests') then 'interests' end,
      case when not can_see(p.id, 'plants') then 'plants' end,
      case when not can_see(p.id, 'pets') then 'pets' end,
      case when not can_see(p.id, 'posts') then 'posts' end], null)
  from profiles p where p.id = p_id;
$$;
grant execute on function public.user_card(uuid) to authenticated;

-- 6. Mina vänner
create or replace function public.my_friends()
returns table (id uuid, display_name text, city text)
language sql stable security definer set search_path = public as $$
  select p.id, p.display_name, case when can_see(p.id, 'city') then p.city end
  from friend_requests f join profiles p on p.id = case when f.from_user = auth.uid() then f.to_user else f.from_user end
  where f.status = 'accepted' and auth.uid() in (f.from_user, f.to_user)
  order by p.display_name;
$$;
grant execute on function public.my_friends() to authenticated;
