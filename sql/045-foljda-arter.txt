-- 045: Följ växtarter. Du får notis när någon ställer en fråga som nämner en art du följer.
create table if not exists public.plant_follows (
  user_id uuid not null references auth.users(id) on delete cascade,
  species text not null check (char_length(species) between 2 and 60),
  created_at timestamptz not null default now(),
  primary key (user_id, species)
);
alter table public.plant_follows enable row level security;
drop policy if exists "egna arter läser" on public.plant_follows;
drop policy if exists "egna arter lägger till" on public.plant_follows;
drop policy if exists "egna arter tar bort" on public.plant_follows;
create policy "egna arter läser" on public.plant_follows for select using (auth.uid() = user_id);
create policy "egna arter lägger till" on public.plant_follows for insert with check (auth.uid() = user_id);
create policy "egna arter tar bort" on public.plant_follows for delete using (auth.uid() = user_id);
