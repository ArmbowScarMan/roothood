-- 092: Växtpassning och bilder får id:n så att växter och namn går att trycka på i appen.
-- sitter_tasks och sitter_beds får växtens id (plant_id), plant_photos får fotografens id (user_id).
-- Kör hela filen en gång i SQL-editorn. Går att köra flera gånger.
drop function if exists public.sitter_tasks();
create function public.sitter_tasks()
returns table(owner_id uuid, owner_name text, plant_row_id bigint, plant_name text, last_watered date, water_interval_days integer, ends_on date, plant_id bigint)
language sql stable security definer set search_path = public as $$
  select ps.owner_id, pr.display_name, up.id, coalesce(up.custom_name, p.swedish_name), up.last_watered,
         coalesce(up.custom_water_interval_days, p.water_interval_days)::int, ps.ends_on, up.plant_id
  from public.plant_sitters ps
  join public.user_plants up on up.user_id = ps.owner_id
  left join public.plants p on p.id = up.plant_id
  left join public.profiles pr on pr.id = ps.owner_id
  where ps.sitter_id = auth.uid() and current_date between ps.starts_on and ps.ends_on
$$;
grant execute on function public.sitter_tasks() to authenticated;

drop function if exists public.sitter_beds();
create function public.sitter_beds()
returns table(owner_id uuid, owner_name text, ends_on date, bed_type text, plant_name text, water_level integer, light_level integer, plant_id bigint)
language sql stable security definer set search_path = public as $$
  select ps.owner_id, pr.display_name, ps.ends_on, a->>'t', pl.swedish_name, pl.water_level, pl.light_level, pl.id
  from plant_sitters ps
  join profiles pr on pr.id = ps.owner_id
  join homes h on h.user_id = ps.owner_id
  cross join lateral jsonb_array_elements(coalesce(h.rooms->'plan'->'areas', '[]'::jsonb)) a
  cross join lateral jsonb_array_elements(coalesce(a->'spots', '[]'::jsonb)) s
  join plants pl on pl.id::text = s->>'pid'
  where ps.sitter_id = auth.uid() and current_date between ps.starts_on and ps.ends_on;
$$;
grant execute on function public.sitter_beds() to authenticated;

drop function if exists public.plant_photos(bigint);
create function public.plant_photos(p_plant_id bigint)
returns table(url text, display_name text, user_id uuid)
language sql stable security definer set search_path = public as $$
  select x.url, x.display_name, x.user_id from (
    select ph.url, pr.display_name, up.user_id, up.created_at, ph.n
    from user_plants up
    join profiles pr on pr.id = up.user_id and pr.share_photos
    cross join lateral unnest(up.photos) with ordinality as ph(url, n)
    where up.plant_id = p_plant_id
      and up.user_id is distinct from auth.uid()
      and ph.url like 'https://%'
      and not exists (select 1 from reports r where r.photo_url = ph.url and r.reporter_id = auth.uid())
      and (select count(distinct r.reporter_id) from reports r where r.photo_url = ph.url) < 3
  ) x
  order by x.created_at desc, x.n desc
  limit 24;
$$;
grant execute on function public.plant_photos(bigint) to authenticated, anon;

-- ===== Förfrågan om passning: passaren accepterar eller tackar nej, och båda kan avsluta =====
alter table public.plant_sitters add column if not exists status text not null default 'pending';
alter table public.plant_sitters add column if not exists responded_at timestamptz;
alter table public.plant_sitters add column if not exists ended_at timestamptz;
alter table public.plant_sitters add column if not exists ended_by uuid;
alter table public.plant_sitters drop constraint if exists plant_sitters_status_check;
alter table public.plant_sitters add constraint plant_sitters_status_check check (status in ('pending', 'accepted', 'declined', 'ended'));

-- Status ändras bara via funktionerna nedan. En ny passning börjar alltid som en förfrågan.
create or replace function public.guard_plant_sitter()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if tg_op = 'INSERT' then
    new.status := 'pending'; new.responded_at := null; new.ended_at := null; new.ended_by := null;
  elsif new.status is distinct from old.status and coalesce(current_setting('roothood.sit', true), '') <> '1' then
    raise exception 'SIT_STATUS';
  end if;
  return new;
end $$;
drop trigger if exists guard_plant_sitter_t on public.plant_sitters;
create trigger guard_plant_sitter_t before insert or update on public.plant_sitters for each row execute function public.guard_plant_sitter();

-- Notis till telefonen (om den finns påslagen): förfrågan till passaren, svar och avslut till den andra
create or replace function public.trg_push_sitter()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if tg_op = 'INSERT' then
    perform public.push_event('sit_req', new.sitter_id, new.owner_id, to_char(new.starts_on, 'DD/MM') || '–' || to_char(new.ends_on, 'DD/MM'));
  elsif new.status is distinct from old.status then
    if new.status = 'accepted' then perform public.push_event('sit_ok', new.owner_id, new.sitter_id, null);
    elsif new.status = 'declined' then perform public.push_event('sit_no', new.owner_id, new.sitter_id, null);
    elsif new.status = 'ended' then perform public.push_event('sit_end', case when new.ended_by = new.owner_id then new.sitter_id else new.owner_id end, new.ended_by, null);
    end if;
  end if;
  return new;
end $$;
drop trigger if exists push_sitter_t on public.plant_sitters;
create trigger push_sitter_t after insert or update on public.plant_sitters for each row execute function public.trg_push_sitter();

-- Passaren svarar på en förfrågan
create or replace function public.sitter_answer(p_id bigint, p_accept boolean)
returns void language plpgsql security definer set search_path = public as $$
begin
  perform set_config('roothood.sit', '1', true);
  update plant_sitters set status = case when p_accept then 'accepted' else 'declined' end, responded_at = now()
  where id = p_id and sitter_id = auth.uid() and status = 'pending';
  if not found then raise exception 'SIT_NOT_FOUND'; end if;
end $$;
grant execute on function public.sitter_answer(bigint, boolean) to authenticated;

-- Ägaren eller passaren avslutar en passning (eller drar tillbaka en förfrågan)
create or replace function public.sitter_end(p_id bigint)
returns void language plpgsql security definer set search_path = public as $$
begin
  perform set_config('roothood.sit', '1', true);
  update plant_sitters set status = 'ended', ended_at = now(), ended_by = auth.uid()
  where id = p_id and (owner_id = auth.uid() or sitter_id = auth.uid()) and status in ('pending', 'accepted');
  if not found then raise exception 'SIT_NOT_FOUND'; end if;
end $$;
grant execute on function public.sitter_end(bigint) to authenticated;

-- Alla passningar där jag är ägare eller passare (pågående, väntande och de senaste 60 dagarna), med namn
create or replace function public.sitter_inbox()
returns table(id bigint, owner_id uuid, owner_name text, sitter_id uuid, sitter_name text, starts_on date, ends_on date, status text, created_at timestamptz, responded_at timestamptz, ended_at timestamptz, ended_by uuid, me_owner boolean)
language sql stable security definer set search_path = public as $$
  select ps.id, ps.owner_id, po.display_name, ps.sitter_id, pse.display_name, ps.starts_on, ps.ends_on, ps.status, ps.created_at, ps.responded_at, ps.ended_at, ps.ended_by, ps.owner_id = auth.uid()
  from plant_sitters ps
  left join profiles po on po.id = ps.owner_id
  left join profiles pse on pse.id = ps.sitter_id
  where (ps.owner_id = auth.uid() or ps.sitter_id = auth.uid())
    and (ps.status in ('pending', 'accepted') or ps.created_at > now() - interval '60 days')
  order by ps.created_at desc
  limit 60;
$$;
grant execute on function public.sitter_inbox() to authenticated;
