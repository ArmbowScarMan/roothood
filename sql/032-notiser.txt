-- Roothood steg 032: notiser i mobilen när du får ett meddelande, en vänförfrågan eller när någon accepterar dig.
-- Kräver steg 029 och att vattningspåminnelserna redan är påslagna (migration 008). Det gör inget om du kör den två gånger.

create extension if not exists pg_net with schema extensions;

create or replace function public.push_event(p_event text, p_to uuid, p_from uuid, p_text text)
returns void language plpgsql security definer set search_path = public, extensions as $$
declare cmd text; bearer text; secret text;
begin
  select command into cmd from cron.job where jobname = 'roothood-paminnelser' limit 1;
  if cmd is null then return; end if;
  bearer := substring(cmd from 'Bearer ([^'']+)');
  secret := substring(cmd from 'x-cron-secret'', ''([^'']+)');
  if bearer is null or secret is null then return; end if;
  perform net.http_post(
    url := 'https://lboomcqxksqahkugpxew.supabase.co/functions/v1/send-reminders',
    headers := jsonb_build_object('Content-Type', 'application/json', 'Authorization', 'Bearer ' || bearer, 'x-cron-secret', secret),
    body := jsonb_build_object('event', p_event, 'to', p_to, 'from', p_from, 'text', left(coalesce(p_text, ''), 120)));
exception when others then
  null; -- en notis får aldrig stoppa ett meddelande
end $$;
revoke all on function public.push_event(text, uuid, uuid, text) from public, anon, authenticated;

create or replace function public.trg_push_message() returns trigger language plpgsql security definer set search_path = public as $$
begin perform public.push_event('message', new.to_user, new.from_user, new.body); return new; end $$;
drop trigger if exists push_message_t on public.messages;
create trigger push_message_t after insert on public.messages for each row execute function public.trg_push_message();

create or replace function public.trg_push_friend() returns trigger language plpgsql security definer set search_path = public as $$
begin
  if tg_op = 'INSERT' then
    perform public.push_event('friend_req', new.to_user, new.from_user, new.message);
  elsif new.status = 'accepted' and old.status is distinct from 'accepted' then
    perform public.push_event('friend_ok', new.from_user, new.to_user, null);
  end if;
  return new;
end $$;
drop trigger if exists push_friend_t on public.friend_requests;
create trigger push_friend_t after insert or update on public.friend_requests for each row execute function public.trg_push_friend();
